<!-- Rule 1: In order to -->
<div class="p-space-md rounded-lg bg-surface-container-low flex flex-col gap-1.5">
<div class="flex items-center justify-between">
<span class="font-label-lg text-label-lg text-primary font-bold">1. In order to + Verb (Infinitive)</span>
<span class="font-label-sm text-label-sm px-2 py-0.5 rounded-full bg-primary-fixed text-on-primary-fixed">Action Focus</span>
</div>
<p class="font-body-md text-body-md text-on-surface-variant">
          Used when the subject performing the main action is the same person or group.
        </p>
<div class="p-2.5 rounded bg-surface-container-lowest text-primary font-body-md text-body-md">
<em>"Thousands of people arrive early <strong>in order to watch</strong> the festive parade in New York City."</em>
</div>
</div>
<!-- Rule 2: So that -->
<div class="p-space-md rounded-lg bg-surface-container-low flex flex-col gap-1.5">
<div class="flex items-center justify-between">
<span class="font-label-lg text-label-lg text-tertiary font-bold">2. So that + Subject + Modal verb</span>
<span class="font-label-sm text-label-sm px-2 py-0.5 rounded-full bg-tertiary-fixed text-on-tertiary-fixed">Outcome Focus</span>
</div>
<p class="font-body-md text-body-md text-on-surface-variant">
          Common when introducing a different subject or explaining the desired result (often paired with <em>can, could, will</em>).
        </p>
<div class="p-2.5 rounded bg-surface-container-lowest text-tertiary font-body-md text-body-md">
<em>"Parents hide chocolate eggs carefully <strong>so that children can enjoy</strong> the morning search."</em>

  let timerInterval = null;
  let timeRemaining = 300; // 5 minutes in seconds
  let isRunning = false;

  function updateDisplay() {
    const minutes = Math.floor(timeRemaining / 60);
    const seconds = timeRemaining % 60;
    const formatted = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
    const display = document.getElementById('timerDisplay');
    if (display) display.textContent = formatted;
  }

  function toggleClassTimer() {
    const btnText = document.getElementById('timerBtnText');
    const btnIcon = document.getElementById('timerIcon');

    if (!isRunning) {
      isRunning = true;
      if (btnText) btnText.textContent = "Pause";
      if (btnIcon) btnIcon.textContent = "pause";
      timerInterval = setInterval(() => {
        if (timeRemaining > 0) {
          timeRemaining--;
          updateDisplay();
        } else {
          clearInterval(timerInterval);
          isRunning = false;
          if (btnText) btnText.textContent = "Time Up!";
          if (btnIcon) btnIcon.textContent = "done";
        }
      }, 1000);
    } else {
      isRunning = false;
      clearInterval(timerInterval);
      if (btnText) btnText.textContent = "Resume";
      if (btnIcon) btnIcon.textContent = "play_arrow";
    }
  }

  function resetClassTimer() {
    clearInterval(timerInterval);
    isRunning = false;
    timeRemaining = 300;
    updateDisplay();
    const btnText = document.getElementById('timerBtnText');
    const btnIcon = document.getElementById('timerIcon');
    if (btnText) btnText.textContent = "Start 5m";
    if (btnIcon) btnIcon.textContent = "play_arrow";
  }

      const utterance = new SpeechSynthesisUtterance(sentence);
      utterance.lang = locale === 'us' ? 'en-US' : 'en-GB';
      utterance.rate = 0.9;
      window.speechSynthesis.cancel();
      window.speechSynthesis.speak(utterance);
    }

  (function() {
    // 1. Objectives drawer toggle
    const toggleBtn = document.getElementById('objectives-toggle');
    const drawer = document.getElementById('objectives-drawer');
    if (toggleBtn && drawer) {
      toggleBtn.addEventListener('click', () => {
        const isHidden = drawer.classList.contains('hidden');
        if (isHidden) {
          drawer.classList.remove('hidden');
          drawer.classList.add('flex');
        } else {
          drawer.classList.add('hidden');
          drawer.classList.remove('flex');
        }
      });
    }

    // 2. Vocabulary pill modal interactive viewer
    const modal = document.getElementById('vocab-modal');
    const vocabCard = document.getElementById('vocab-card');
    const termEl = document.getElementById('vocab-term');
    const phoneticEl = document.getElementById('vocab-phonetic');
    const defEl = document.getElementById('vocab-def');
    const regionEl = document.getElementById('vocab-region');
    const closeBtn = document.getElementById('close-vocab-btn');
    const dismissBtn = document.getElementById('dismiss-vocab-btn');

    function openVocab(term, phonetic, def, region) {
      termEl.textContent = term;
      phoneticEl.textContent = phonetic;
      defEl.textContent = def;
      regionEl.textContent = region;
      modal.classList.remove('hidden');
      modal.classList.add('flex');
      setTimeout(() => {
        vocabCard.classList.remove('scale-95', 'opacity-0');
        vocabCard.classList.add('scale-100', 'opacity-100');
      }, 10);
    }

    function closeVocab() {
      vocabCard.classList.remove('scale-100', 'opacity-100');
      vocabCard.classList.add('scale-95', 'opacity-0');
      setTimeout(() => {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
      }, 150);
    }

    document.querySelectorAll('.vocab-btn').forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        const term = btn.getAttribute('data-term');
        const phonetic = btn.getAttribute('data-phonetic');
        const def = btn.getAttribute('data-def');
        const region = btn.getAttribute('data-region');
        openVocab(term, phonetic, def, region);
      });
    });

    if (closeBtn) closeBtn.addEventListener('click', closeVocab);
    if (dismissBtn) dismissBtn.addEventListener('click', closeVocab);
    if (modal) {
      modal.addEventListener('click', (e) => {
        if (e.target === modal) closeVocab();
      });
    }

    // 3. Speech starter quick copy
    const toast = document.getElementById('speech-toast');
    document.querySelectorAll('.speech-starter-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        const text = btn.getAttribute('data-starter');
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(text).catch(() => {});
        }
        if (toast) {
          toast.classList.remove('hidden');
          toast.classList.add('flex');
          setTimeout(() => {
            toast.classList.add('hidden');
            toast.classList.remove('flex');
          }, 2000);
        }
      });
    });
  })();
